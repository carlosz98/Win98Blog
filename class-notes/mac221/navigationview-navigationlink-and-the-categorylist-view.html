NavigationView, NavigationLink and the CategoryList View

NavigationView adds a navigation bar and a back button; NavigationLink is a tappable view that pushes a new screen. Put the title modifier on the content inside the NavigationView, not on the NavigationView itself.

Basics

NavigationView {
    NavigationLink(destination: Text("Second View")) {   // where to go
        Text("Hello MAC221")                               // what you tap (can be an Image)
    }
    .navigationBarTitle("SwiftUI Navigation")             // displayMode: .automatic, .large or .inline
}

.navigationTitle("...") is the newer name for the same thing. An image can be the link label too: Image("ocean").renderingMode(.original).resizable().scaledToFit().clipShape(Circle()).

Passing data to the next screen (coin flip)

struct ResultView: View {
    var choice: String                         // filled in by whoever opens this view
    var body: some View { Text("You chose \(choice)") }
}

struct ContentView: View {
    var body: some View {
        NavigationView {
            VStack(spacing: 30) {
                Text("You're going to flip a coin. Heads or tails?")
                NavigationLink(destination: ResultView(choice: "Heads")) { Text("Choose Heads") }
                NavigationLink(destination: ResultView(choice: "Tails")) { Text("Choose Tails") }
            }
            .navigationTitle("Navigation")
        }
    }
}

Navigating from a button (programmatic): a @State var linkIsActive = false, a Button that sets it to true, and NavigationLink(destination: SecondView(), isActive: $linkIsActive) { EmptyView() }.

CategoryList View

The CategoryList lecture builds up in four steps.

1. A List of strings. id: \.self is needed because a String isn't Identifiable.

struct CategoryListView: View {
    let categories = ["Category 1", "Category 2", "Category 3"]
    var body: some View {
        List(categories, id: \.self) { category in
            Text(category)
        }
    }
}

2. Make it the app's first screen in the @main App file:

@main
struct CategoryApp: App {
    var body: some Scene {
        WindowGroup { CategoryListView() }
    }
}

3. Wrap it in a NavigationView and make each row a NavigationLink to a detail view.

4. Use a model struct with an image:

struct Category: Identifiable {
    let id = UUID()                 // unique id, so List doesn't need id: \.self
    let name: String
    let description: String
    let image: String               // asset name
}

struct CategoryListView: View {
    let categories = [
        Category(name: "Category 1", description: "Description 1", image: "category1"),
        Category(name: "Category 2", description: "Description 2", image: "category2"),
    ]

    var body: some View {
        NavigationView {
            List(categories) { category in
                NavigationLink(destination: Text("Detail View for \(category.name)")) {
                    HStack {
                        Image(category.image).resizable().frame(width: 30, height: 30)
                        Text(category.name)
                    }
                }
            }
            .navigationBarTitle("Categories")
        }
    }
}

Dynamic List: add and delete rows

The last List example keeps the data in an ObservableObject store (see the next section) so rows can be added and removed.

struct Person: Identifiable {
    let id = UUID()
    let name: String
}

class PeopleStore: ObservableObject {
    @Published var people = [Person(name: "John"), Person(name: "Alice"), Person(name: "Bob")]

    func addPerson(_ person: Person) { people.append(person) }
    func removePerson(_ person: Person) {
        if let index = people.firstIndex(where: { $0.id == person.id }) {
            people.remove(at: index)
        }
    }
}

struct ContentView: View {
    @ObservedObject var peopleStore = PeopleStore()
    @State private var newPersonName = ""

    var body: some View {
        NavigationView {
            List {
                Section(header: Text("Add Person")) {
                    HStack {
                        TextField("Name", text: $newPersonName)
                        Button("Add", action: addPerson)
                    }
                }
                Section(header: Text("People")) {
                    ForEach(peopleStore.people) { person in
                        Text(person.name)
                    }
                    .onDelete(perform: removePeople)       // swipe to delete
                }
            }
            .listStyle(GroupedListStyle())
            .navigationBarTitle("Dynamic List")
        }
    }

    func addPerson() {
        peopleStore.addPerson(Person(name: newPersonName))
        newPersonName = ""                                 // clear the text field
    }

    func removePeople(at offsets: IndexSet) {
        for index in offsets { peopleStore.removePerson(peopleStore.people[index]) }
    }
}

  • .onDelete only works on ForEach, not directly on List(items).

  • List styles: PlainListStyle(), GroupedListStyle(), InsetGroupedListStyle().
